func main() {
	inputs := append([]string{}, curated...)
	for a := 'a'; a <= 'z'; a++ {
		for b := 'a'; b <= 'z'; b++ {
			inputs = append(inputs, string([]rune{a, b}))
		}
	}
	r := rand.New(rand.NewSource(7))
	alpha := "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_"
	for i := 0; i < 2000; i++ {
		n := r.Intn(16)
		b := make([]byte, n)
		for k := range b {
			b[k] = alpha[r.Intn(len(alpha))]
		}
		inputs = append(inputs, string(b))
	}
	// Structured random tags: lang[-script][-region][-variant][-ext].
	langs := []string{"de", "en", "zh", "pt", "cs", "fr", "es", "ja", "ko", "pl", "ru", "tr", "it", "sr", "no", "nb", "iw", "sh", "mo", "und", "cmn", "yue", "xx", "qaa", "tlh", "fil", "tl"}
	scripts := []string{"", "Latn", "Hans", "Hant", "Cyrl", "Arab", "Qaai", "Zzzz", "Xxxx", "Jpan", "Kore"}
	regions := []string{"", "DE", "US", "CN", "TW", "HK", "BR", "PT", "CZ", "419", "001", "150", "ZZ", "XK", "QO", "AT", "CH", "MD", "AA", "999"}
	tails := []string{"", "-1996", "-1901", "-x-foo", "-u-co-phonebk", "-u-rg-uszzzz", "-u-rg-dezzzz", "-t-en", "-a-bcd", "-whoops", "-valencia"}
	for i := 0; i < 1500; i++ {
		s := langs[r.Intn(len(langs))]
		if sc := scripts[r.Intn(len(scripts))]; sc != "" {
			s += "-" + sc
		}
		if rg := regions[r.Intn(len(regions))]; rg != "" {
			s += "-" + rg
		}
		s += tails[r.Intn(len(tails))]
		inputs = append(inputs, s)
	}
	enc := json.NewEncoder(os.Stdout)
	for _, s := range inputs {
		tag, err := language.Parse(s)
		idx, conf := -1, "-"
		if tag != language.Und {
			_, i, c := matcher.Match(tag)
			idx, conf = i, c.String()
		}
		if err := enc.Encode(struct {
			In    string `json:"in"`
			Ok    bool   `json:"ok"`
			Tag   string `json:"tag"`
			Und   bool   `json:"und"`
			Index int    `json:"index"`
			Conf  string `json:"conf"`
		}{s, err == nil, tag.String(), tag == language.Und, idx, conf}); err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(1)
		}
	}
}
